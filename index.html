The 10-Line CSS Cutout Secret

badge-cutout.css 
/* 1. Host container must be relative */
.avatar-wrap {
  position: relative;
  display: inline-flex;
}

/* 2. Absolute badge with 35% corner offset */
.badge {
  position: absolute;
  top: 0;
  right: 0;
  transform: translate(35%, -35%);
  border-radius: 9999px;
  background: #ef4444;
  color: #ffffff;
  padding: 2px 6px;
  font-size: 11px;
  font-weight: 700;

  /* 3. The Cutout Magic: border matches parent bg */
  border: 2px solid var(--bg-card);
  box-sizing: content-box;
}
